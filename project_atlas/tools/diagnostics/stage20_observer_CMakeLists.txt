cmake_minimum_required(VERSION 3.16)
project(atlas_stage20_observer)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(robot_localization REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(Eigen3 REQUIRED)
foreach(dep angles tf2 tf2_ros tf2_geometry_msgs geometry_msgs nav_msgs sensor_msgs std_srvs diagnostic_msgs diagnostic_updater)
  find_package(${dep} REQUIRED)
endforeach()
add_executable(stage20_ekf_observer stage20_ekf_observer.cpp)
target_link_libraries(stage20_ekf_observer Eigen3::Eigen)
find_library(ATLAS_RL_LIBRARY rl_lib PATHS /opt/ros/humble/lib REQUIRED)
target_link_libraries(stage20_ekf_observer ${ATLAS_RL_LIBRARY})
ament_target_dependencies(stage20_ekf_observer robot_localization rclcpp std_msgs angles tf2 tf2_ros tf2_geometry_msgs geometry_msgs nav_msgs sensor_msgs std_srvs diagnostic_msgs diagnostic_updater)
